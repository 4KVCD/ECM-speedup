# library: the GPU (Vulkan compute) decoder
set( LIB_NAME VulkanLib )

file( GLOB SRC_FILES "*.cpp" )
file( GLOB INC_FILES "*.h" )
file( GLOB SHADER_SRCS "shaders/*.comp" )
file( GLOB SHADER_INCS "shaders/*.glsl" )

# Every shader is compiled to SPIR-V and embedded as a uint32_t array; VkShaderTable.cpp lists them by name.
set( SPV_DIR ${CMAKE_CURRENT_BINARY_DIR}/spv )
file( MAKE_DIRECTORY ${SPV_DIR} )
set( SPV_HEADERS )
set( SHADER_TABLE_INCLUDES "" )
set( SHADER_TABLE_ENTRIES "" )
foreach( src ${SHADER_SRCS} )
  get_filename_component( name ${src} NAME_WE )
  set( out ${SPV_DIR}/${name}.h )
  add_custom_command( OUTPUT ${out}
                      COMMAND ${GLSLANG_EXE} -V --target-env vulkan1.3 --quiet --vn spv_${name}
                              -I${CMAKE_CURRENT_SOURCE_DIR}/shaders -o ${out} ${src}
                      DEPENDS ${src} ${SHADER_INCS}
                      COMMENT "glslang ${name}.comp" VERBATIM )
  list( APPEND SPV_HEADERS ${out} )
  string( APPEND SHADER_TABLE_INCLUDES "#include \"${name}.h\"\n" )
  string( APPEND SHADER_TABLE_ENTRIES "  { \"${name}\", spv_${name}, sizeof( spv_${name} ) },\n" )
endforeach()
set( SHADER_TABLE ${SPV_DIR}/VkShaderTable.cpp )
file( GENERATE OUTPUT ${SHADER_TABLE} CONTENT
"// generated by source/Lib/VulkanLib/CMakeLists.txt
#include <cstdint>
#include <cstring>
#include \"VkShaders.h\"
${SHADER_TABLE_INCLUDES}
namespace vkd
{
static const ShaderCode s_shaders[] = {
${SHADER_TABLE_ENTRIES}  { nullptr, nullptr, 0 }
};
const ShaderCode* findShader( const char* name )
{
  for( const ShaderCode* s = s_shaders; s->name; s++ )
  {
    if( strcmp( s->name, name ) == 0 )
    {
      return s;
    }
  }
  return nullptr;
}
}
" )

add_library( ${LIB_NAME} STATIC ${SRC_FILES} ${INC_FILES} ${SHADER_TABLE} ${SPV_HEADERS} ${SHADER_SRCS} ${SHADER_INCS} )
target_include_directories( ${LIB_NAME} PUBLIC . ${VULKAN_HEADERS_DIR} )
target_include_directories( ${LIB_NAME} PRIVATE ${SPV_DIR} )
target_compile_definitions( ${LIB_NAME} PUBLIC ECM_VULKAN=1 )
target_link_libraries( ${LIB_NAME} CommonLib )
if( NOT WIN32 )
  target_link_libraries( ${LIB_NAME} ${CMAKE_DL_LIBS} )
endif()
set_target_properties( ${LIB_NAME} PROPERTIES FOLDER lib )
